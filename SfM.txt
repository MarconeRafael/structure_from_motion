    // Função para encontrar os K vizinhos mais próximos entre dois conjuntos de descritores
    FUNÇÃO KNNMatch(flann, descritores1, descritores2, k)
        correspondencias <- LISTA_VAZIA

        // Para cada descritor na primeira imagem
        PARA CADA descritor1 EM descritores1
            vizinhos <- LISTA_VAZIA

            // Para cada descritor na segunda imagem
            PARA CADA descritor2 EM descritores2
                // Calcular a distância entre os descritores (exemplo: distância Euclidiana)
                distancia <- CalcularDistancia(descritor1, descritor2)

                // Adicionar à lista de vizinhos
                ADICIONAR (distancia, descritor2) A vizinhos
            FIM PARA

            // Ordenar vizinhos pela menor distância
            ORDENAR vizinhos POR distancia ASCENDENTE

            // Selecionar os K vizinhos mais próximos
            melhoresVizinhos <- PRIMEIROS_K_ELEMENTOS(vizinhos, k)

            // Adicionar ao conjunto final de correspondências
            ADICIONAR melhoresVizinhos A correspondencias
        FIM PARA

        RETORNAR correspondencias
    FIM FUNÇÃO

    // Encontrar correspondências entre os descritores das imagens esquerda e direita
    matches <- KNNMatch(flann, descriptorsLeft, descriptorsRight, 2)

    PARA CADA (m, n) EM matches

    FIM PARA

    img1 <- ConverterParaColorido(img1)
    img2 <- ConverterParaColorido(img2)

        // Gerar uma cor aleatória
        cor <- GerarCorAleatoria()

        // Calcular as coordenadas de interseção da linha com as bordas da imagem
        (x0, y0) <- CalcularPontoDeInterseção(0, linha)  // Interseção com a borda esquerda
        (x1, y1) <- CalcularPontoDeInterseção(c, linha)  // Interseção com a borda direita

        // Desenhar a linha na imagem 1

        // Desenhar o ponto de correspondência na imagem 2

    RETORNAR img1, img2
FIM FUNÇÃO
